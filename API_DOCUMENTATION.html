Introduction

L'API ForexPlatform permet à n'importe quelle application (mobile, web, desktop) de récupérer des taux de change en temps réel et d'effectuer des conversions entre 33 devises internationales.


33
Devises supportées
15
Endpoints disponibles
5min
Cache temps réel
✅Les taux sont récupérés en temps réel depuis la Banque Centrale Européenne (BCE). Aucune donnée statique ou fictive.

Comment ça fonctionne

# L'app envoie une requête avec la clé API
App Mobile  ──►  X-API-KEY: fxp_xxx  ──►  ForexPlatform API
                                               ──►  BCE (taux temps réel)
App Mobile  ◄──  {"market_rate": "1.17"}  ◄──  Réponse JSON

Conversion de Montant

⭐C'est l'endpoint principal de l'application. Il applique le taux en temps réel + la commission selon le tier.
POST /api/v1/convert/ Auth: X-API-KEY ou Bearer

Headers requis

HeaderValeur
X-API-KEYfxp_oNGqt67mlKz40hl0Me748a1rQQwMD0qrjVB1pL16xOY
Content-Typeapplication/json

Body — Paramètres

ChampTypeObligatoireDescription
from_currencystringRequisDevise source (ex: "EUR")
to_currencystringRequisDevise cible (ex: "USD")
amountnumberRequisMontant à convertir (doit être > 0)

Exemple de requête

{
  "from_currency": "EUR",
  "to_currency": "USD",
  "amount": 500
}

Réponse 200 OK

{
  "success": true,
  "conversion": {
    "from_currency": "EUR",
    "to_currency":   "USD",
    "amount":         "500",
    "converted_amount":"595.5300",   ← montant final reçu
    "market_rate":    "1.1700",     ← taux brut BCE
    "business_rate":  "1.1911",     ← taux appliqué (+ commission)
    "spread_pct":     "1.50",       ← commission en %
    "fee_amount":     "8.9330",     ← frais prélevés
    "tier":           "standard",
    "source":         "live",       ← taux en temps réel
    "timestamp":      "2026-05-13T18:00:00Z"
  }
}

3. Service API complet

// src/services/forexService.js
import axios from 'axios';
import { API_CONFIG, API_HEADERS } from '../config/api';

const api = axios.create({
  baseURL: API_CONFIG.BASE_URL,
  headers: API_HEADERS,
  timeout: 10000, // 10 secondes max
});

// ✅ Vérifier que l'API fonctionne
export const checkHealth = async () => {
  const { data } = await api.get('/health/');
  return data;
};

// ✅ Récupérer toutes les devises
export const getCurrencies = async () => {
  const { data } = await api.get('/currencies/');
  return data.currencies; // tableau de 33 devises
};

// ✅ Récupérer le taux entre deux devises (temps réel)
export const getRate = async (fromCurrency, toCurrency) => {
  const { data } = await api.get(`/rates/${fromCurrency}/${toCurrency}/`);
  return data.rate; // { pair, market_rate, timestamp }
};

// ✅ Convertir un montant
export const convertAmount = async (fromCurrency, toCurrency, amount) => {
  const { data } = await api.post('/convert/', {
    from_currency: fromCurrency,
    to_currency:   toCurrency,
    amount:        amount,
  });
  return data.conversion; // { converted_amount, market_rate, fee_amount, ... }
};

// ✅ Login (pour wallets et transferts)
export const login = async (username, password) => {
  const { data } = await api.post('/auth/token/', { username, password });
  return data; // { access, refresh }
};

// ✅ Mes wallets (nécessite JWT)
export const getWallets = async (jwtToken) => {
  const { data } = await api.get('/wallets/', {
    headers: { ...API_HEADERS, Authorization: `Bearer ${jwtToken}` }
  });
  return data.wallets;
};

4. Écran de conversion — Exemple complet

// src/screens/ConvertScreen.js
import React, { useState, useEffect } from 'react';
import { View, Text, TextInput, TouchableOpacity,
         ActivityIndicator, StyleSheet, Alert } from 'react-native';
import { getRate, convertAmount } from '../services/forexService';

export default function ConvertScreen() {
  const [from, setFrom]     = useState('EUR');
  const [to, setTo]         = useState('USD');
  const [amount, setAmount] = useState('');
  const [result, setResult] = useState(null);
  const [rate, setRate]     = useState(null);
  const [loading, setLoading] = useState(false);

  // Charger le taux au démarrage
  useEffect(() => {
    loadRate();
  }, [from, to]);

  const loadRate = async () => {
    try {
      const data = await getRate(from, to);
      setRate(data.market_rate);
    } catch (e) {
      Alert.alert('Erreur', 'Impossible de récupérer le taux');
    }
  };

  const handleConvert = async () => {
    if (!amount || isNaN(amount) || Number(amount) <= 0) {
      Alert.alert('Erreur', 'Entrez un montant valide');
      return;
    }
    setLoading(true);
    try {
      const data = await convertAmount(from, to, Number(amount));
      setResult(data);
    } catch (e) {
      Alert.alert('Erreur', e.response?.data?.error || 'Erreur de conversion');
    } finally {
      setLoading(false);
    }
  };

  return (
    <View style={styles.container}>
      <Text style={styles.title}>Convertisseur Forex</Text>

      {rate && (
        <Text style={styles.rateText}>
          Taux {from}/{to} : {rate} (temps réel)
        </Text>
      )}

      <TextInput
        style={styles.input}
        placeholder={`Montant en ${from}`}
        keyboardType="numeric"
        value={amount}
        onChangeText={setAmount}
      />

      <TouchableOpacity style={styles.btn} onPress={handleConvert}>
        {loading
          ? <ActivityIndicator color="#fff" />
          : <Text style={styles.btnText}>Convertir en {to}</Text>}
      </TouchableOpacity>

      {result && (
        <View style={styles.result}>
          <Text style={styles.resultAmount}>{result.converted_amount} {to}</Text>
          <Text style={styles.resultDetail}>Taux : {result.market_rate}</Text>
          <Text style={styles.resultDetail}>Frais : {result.fee_amount} {from}</Text>
        </View>
      )}
    </View>
  );
}

const styles = StyleSheet.create({
  container:    { flex: 1, padding: 20, backgroundColor: '#f8fafc' },
  title:        { fontSize: 24, fontWeight: 'bold', color: '#0f3460', marginBottom: 20 },
  rateText:     { color: '#10b981', marginBottom: 16, fontSize: 13 },
  input:        { borderWidth: 1, borderColor: '#e2e8f0', borderRadius: 8,
                  padding: 12, fontSize: 16, backgroundColor: 'white', marginBottom: 12 },
  btn:          { backgroundColor: '#0f3460', borderRadius: 8, padding: 14, alignItems: 'center' },
  btnText:      { color: 'white', fontWeight: 'bold', fontSize: 16 },
  result:       { marginTop: 24, backgroundColor: 'white', borderRadius: 10, padding: 20 },
  resultAmount: { fontSize: 28, fontWeight: '800', color: '#0f3460' },

5. Gestion des erreurs réseau

// Intercepteur global dans src/services/forexService.js
api.interceptors.response.use(
  response => response,
  error => {
    if (!error.response) {
      // Pas de connexion internet
      throw new Error('Vérifiez votre connexion internet');
    }
    switch (error.response.status) {
      case 403: throw new Error('Clé API invalide');
      case 404: throw new Error('Devise non supportée');
      case 429: throw new Error('Quota dépassé — réessayez dans 1 heure');
      case 503: throw new Error('Service temporairement indisponible');
      default:  throw new Error(error.response.data?.error || 'Erreur inconnue');
    }
  }
);
